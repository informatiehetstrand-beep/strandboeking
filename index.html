const express = require('express'); const cors = require('cors'); const { createClient } = require('@supabase/supabase-js'); const { createMollieClient } = require('@mollie/api-client'); const nodemailer = require('nodemailer'); require('dotenv').config(); const app = express(); app.use(cors({ origin: process.env.FRONTEND_URL || '*' })); app.use(express.json()); app.use(express.urlencoded({ extended: true })); // ─── CLIENTS ───────────────────────────────────────────────────────────────── const supabase = createClient( process.env.SUPABASE_URL, process.env.SUPABASE_KEY ); const mollie = createMollieClient({ apiKey: process.env.MOLLIE_API_KEY }); const mailer = nodemailer.createTransport({ host: process.env.SMTP_HOST || 'mail.strandbedhuren.nl', port: 587, secure: false, auth: { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS } }); // ─── HULPFUNCTIES ──────────────────────────────────────────────────────────── function genereerReserveringNr() { return 'RES-' + Date.now().toString(36).toUpperCase(); } function formatDatum(datum) { const [jaar, maand, dag] = datum.split('-'); const maanden = ['januari','februari','maart','april','mei','juni', 'juli','augustus','september','oktober','november','december']; return `${parseInt(dag)} ${maanden[parseInt(maand)-1]} ${jaar}`; } function bevestigingsmail(reservering) { const productenLijst = reservering.producten .map(p => ` ${p.naam} × ${p.aantal} €${(p.prijs * p.aantal).toFixed(2)} `) .join(''); return `

🌊 Reservering bevestigd!

Jouw stranddag staat vast

Beste ${reservering.naam},

Je reservering is bevestigd en je betaling is ontvangen. We kijken ernaar uit je te verwelkomen op het strand!

Reserveringsnummer ${reservering.reservering_nr}
Datum ${formatDatum(reservering.datum)}
Betaalmethode ${reservering.betaal_methode || 'Online betaald'}
${productenLijst}
Product Bedrag
Totaal betaald €${parseFloat(reservering.totaal).toFixed(2)}

📍 Laat dit e-mailadres of je reserveringsnummer zien bij aankomst.

Tot ziens op het strand! 🏖️
StrandBedHuren.nl

Je ontvangt deze mail omdat je een reservering hebt gemaakt via strandbedhuren.nl

`; } // ─── ROUTES ────────────────────────────────────────────────────────────────── app.get('/', (req, res) => { res.json({ status: 'ok', message: 'Strandboeking API draait!' }); }); app.get('/api/prijzen', async (req, res) => { try { const { data, error } = await supabase.from('prijzen').select('*'); if (error) throw error; res.json(data); } catch (err) { res.status(500).json({ error: 'Kon prijzen niet ophalen' }); } }); app.get('/api/reserveringen', async (req, res) => { try { const { data, error } = await supabase .from('reserveringen') .select('*') .order('aangemaakt_op', { ascending: false }); if (error) throw error; res.json(data); } catch (err) { res.status(500).json({ error: 'Kon reserveringen niet ophalen' }); } }); app.post('/api/betaling', async (req, res) => { try { const { datum, producten, klant, methode } = req.body; if (!datum || !producten || !klant || !klant.naam || !klant.email) { return res.status(400).json({ error: 'Verplichte velden ontbreken' }); } const totaal = producten.reduce((s, p) => s + (p.prijs * p.aantal), 0); const nr = genereerReserveringNr(); await supabase.from('reserveringen').insert({ reservering_nr: nr, naam: klant.naam, email: klant.email, telefoon: klant.telefoon || '', datum, producten, totaal, status: 'pending' }); const payment = await mollie.payments.create({ amount: { currency: 'EUR', value: totaal.toFixed(2) }, description: `Strandboeking ${nr} — ${formatDatum(datum)}`, redirectUrl: `${process.env.FRONTEND_URL}/bevestiging?nr=${nr}`, webhookUrl: `${process.env.API_URL}/api/webhook`, method: methode || undefined, metadata: { reservering_nr: nr, klant_email: klant.email, klant_naam: klant.naam } }); await supabase.from('reserveringen') .update({ mollie_payment_id: payment.id }) .eq('reservering_nr', nr); res.json({ checkoutUrl: payment._links.checkout.href, nr }); } catch (err) { console.error('Betaling fout:', err); res.status(500).json({ error: 'Betaling kon niet worden gestart' }); } }); app.post('/api/webhook', async (req, res) => { try { const { id } = req.body; if (!id) return res.status(400).send('Geen payment ID'); const payment = await mollie.payments.get(id); const nr = payment.metadata?.reservering_nr; if (!nr) return res.status(400).send('Geen reserveringsnummer'); if (payment.status === 'paid') { await supabase.from('reserveringen') .update({ status: 'confirmed', betaal_methode: payment.method }) .eq('reservering_nr', nr); const { data: reservering } = await supabase .from('reserveringen').select('*').eq('reservering_nr', nr).single(); if (reservering) { await mailer.sendMail({ from: `StrandBedHuren.nl <${process.env.SMTP_USER}>`, to: reservering.email, subject: `✅ Reservering bevestigd — ${nr}`, html: bevestigingsmail(reservering) }); } } else if (['expired','failed','canceled'].includes(payment.status)) { await supabase.from('reserveringen') .update({ status: 'cancelled' }) .eq('reservering_nr', nr); } res.status(200).send('OK'); } catch (err) { console.error('Webhook fout:', err); res.status(500).send('Webhook fout'); } }); app.patch('/api/reserveringen/:nr/annuleer', async (req, res) => { try { const { nr } = req.params; const { error } = await supabase.from('reserveringen') .update({ status: 'cancelled' }).eq('reservering_nr', nr); if (error) throw error; res.json({ success: true }); } catch (err) { res.status(500).json({ error: 'Kon niet annuleren' }); } }); app.patch('/api/prijzen/:id', async (req, res) => { try { const { id } = req.params; const { prijs } = req.body; const { error } = await supabase.from('prijzen') .update({ prijs }).eq('product_id', id); if (error) throw error; res.json({ success: true }); } catch (err) { res.status(500).json({ error: 'Kon prijs niet bijwerken' }); } }); const PORT = process.env.PORT || 3001; app.listen(PORT, () => console.log(`Strandboeking API draait op poort ${PORT}`)); module.exports = app;
